cmake_minimum_required(VERSION 3.20)

file(STRINGS ${CMAKE_CURRENT_SOURCE_DIR}/VERSION QUANT_PROJECT_VERSION
  LIMIT_COUNT 1)
project(modern_cpp_for_quant VERSION ${QUANT_PROJECT_VERSION} LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

option(QUANT_BUILD_CHAPTERS "Build chapter and appendix examples" ON)
option(QUANT_BUILD_BACKTEST "Build the final event-driven backtest" ON)
option(QUANT_BUILD_CAPSTONES "Build the factor and order-book capstones" ON)
option(QUANT_BUILD_NEGATIVE_TESTS "Enable intentional-failure tests" ON)
option(QUANT_BUILD_PYTHON "Build optional Python extension modules" ON)
option(QUANT_BUILD_FUZZERS "Build Clang/libFuzzer parser targets" OFF)

include(CTest)

  add_subdirectory(code/ch01)

  add_subdirectory(code/ch03)

  add_subdirectory(code/ch06)

  add_subdirectory(code/ch08)
  add_subdirectory(code/ch09)
  add_subdirectory(code/ch10)
  add_subdirectory(code/ch11)
  add_subdirectory(code/ch12)
  add_subdirectory(code/ch13)
  add_subdirectory(code/ch14)
  add_subdirectory(code/ch15)
  set(CH16_BUILD_PYTHON ${QUANT_BUILD_PYTHON})
  add_subdirectory(code/ch16)
endif()

if(QUANT_BUILD_BACKTEST)
  add_subdirectory(project)
endif()

if(QUANT_BUILD_CAPSTONES)

  add_subdirectory(capstones/factor_kernel)
